    function levenshteinDistance(a, b) {
      if (a.length < b.length) [a, b] = [b, a];
      if (b.length === 0) return a.length;
      let prevRow = Array.from({ length: b.length + 1 }, (_, i) => i);
      for (let i = 0; i < a.length; i++) {
        let currRow = [i + 1];
        for (let j = 0; j < b.length; j++) {
          const insertions = prevRow[j + 1] + 1;
          const deletions = currRow[j] + 1;
          const substitutions = prevRow[j] + (a[i] !== b[j] ? 1 : 0);
          currRow.push(Math.min(insertions, deletions, substitutions));
        }
        prevRow = currRow;
      }
      return prevRow[b.length];
    }

    function getComparisonKey(str, isDepartamento = false) {
      let s = cleanString(str).toLowerCase();
      if (isDepartamento) {
        s = s.replace(/^(departamento\s+de\s+la\s+carrera\s+de\s+|departamento\s+de\s+carrera\s+de\s+|departamento\s+de\s+|departarmento\s+de\s+|departamento\s+del\s+|departamento\s+en\s+|departamento\s+|depto\.?\s+de\s+|dep\.?\s+de\s+|depto\.?\s+|dep\.?\s+|carrera\s+de\s+|carrera\s+|escuela\s+de\s+|escuela\s+)/i, "");
      }
      return s.normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-z0-9\s]/g, " ").replace(/\s+/g, " ").trim();
    }

    function countAccents(str) { return (str.match(/[áéíóúÁÉÍÓÚñÑüÜ]/g) || []).length; }

    function unifyCanonicalValues(records) {
      const fieldsToUnify = [
        { name: "Departamento", isDepto: true, fuzzy: true },
        { name: "Carrera", isDepto: false, fuzzy: true },
        { name: "Facultad", isDepto: false, fuzzy: false },
        { name: "Centro_Regional", isDepto: false, fuzzy: false },
        { name: "Direccion", isDepto: false, fuzzy: true },
        { name: "Tipo_Contratacion", isDepto: false, fuzzy: false },
        { name: "Jornada", isDepto: false, fuzzy: false },
        { name: "Nombre_Capacitacion", isDepto: false, fuzzy: false }
      ];

      fieldsToUnify.forEach(({ name, isDepto, fuzzy }) => {
        const keyFreq = new Map();
        const groups = new Map();

        records.forEach(r => {
          const val = r[name];
          if (!val) return;
          const key = getComparisonKey(val, isDepto);
          if (!key) return;

          keyFreq.set(key, (keyFreq.get(key) || 0) + 1);
          if (!groups.has(key)) groups.set(key, new Map());
          const variantsMap = groups.get(key);
          variantsMap.set(val, (variantsMap.get(val) || 0) + 1);
        });

        const sortedKeys = Array.from(keyFreq.keys()).sort((a, b) => keyFreq.get(b) - keyFreq.get(a));
        const clusterMap = new Map();
        const canonicalClusters = [];

        sortedKeys.forEach(k => {
          let matched = false;
          if (fuzzy && k.length >= 7) {
            for (const canon of canonicalClusters) {
              if (canon.length >= 7 && Math.abs(k.length - canon.length) <= 2) {
                const dist = levenshteinDistance(k, canon);
                const maxLen = Math.max(k.length, canon.length);
                if (dist <= 2 && (1 - dist / maxLen) >= 0.84) {
                  clusterMap.set(k, canon);
                  const targetVariants = groups.get(canon);
                  groups.get(k).forEach((cnt, form) => {
                    targetVariants.set(form, (targetVariants.get(form) || 0) + cnt);
                  });
                  matched = true;
                  break;
                }
              }
            }
          }
          if (!matched) {
            canonicalClusters.push(k);
            clusterMap.set(k, k);
          }
        });

        const canonicalLabel = new Map();
        canonicalClusters.forEach(canonKey => {
          const variantsMap = groups.get(canonKey);
          let bestForm = "";
          let bestScore = -1;
          variantsMap.forEach((count, form) => {
            const score = (countAccents(form) > 0 ? 100000 : 0) + count;
            if (score > bestScore) {
              bestScore = score;
              bestForm = form;
            }
          });
          canonicalLabel.set(canonKey, bestForm);
        });

        records.forEach(r => {
          const key = getComparisonKey(r[name], isDepto);
          const targetCluster = clusterMap.get(key);
          if (targetCluster && canonicalLabel.has(targetCluster)) {
            r[name] = canonicalLabel.get(targetCluster);
          }
        });
      });

      return records;
    }

    function normalizeDireccion(rawVal) {
      let s = cleanString(rawVal);
      if (!s) return "Sin especificar";
      
      const cleanLower = stripAccents(s).toLowerCase().replace(/[^a-z0-9\s]/g, " ").replace(/\s+/g, " ").trim();
      
      if (/\b(dipp)\b/.test(cleanLower) || (cleanLower.includes("innovacion") && cleanLower.includes("produccion"))) {
        return "Dirección de Innovación y Producción Pedagógica (DIPP)";
      }
      if (/\b(die)\b/.test(cleanLower) || (cleanLower.includes("innovacion") && cleanLower.includes("educativa"))) {
        return "Dirección de Innovación Educativa (DIE)";
      }
      if (/\b(sedp)\b/.test(cleanLower) || cleanLower.includes("desarrollo de personal") || cleanLower.includes("recursos humanos")) {
        return "Secretaría Ejecutiva de Desarrollo de Personal (SEDP)";
      }
      if (/\b(degt)\b/.test(cleanLower) || cleanLower.includes("gestion de tecnologia") || cleanLower.includes("tecnologia")) {
        return "Dirección Ejecutiva de Gestión de Tecnología (DEGT)";
      }
      if (/\b(voae)\b/.test(cleanLower) || cleanLower.includes("orientacion y asuntos estudiantiles")) {
        return "Vicerrectoría de Orientación y Asuntos Estudiantiles (VOAE)";
      }
      if (/\b(vra)\b/.test(cleanLower) || (cleanLower.includes("vicerrectoria") && cleanLower.includes("academica"))) {
        return "Vicerrectoría Académica (VRA)";
      }
      if (/\b(dda)\b/.test(cleanLower) || (cleanLower.includes("direccion") && cleanLower.includes("docencia"))) {
        return "Dirección de Docencia (DDA)";
      }
      if (/\b(dicu)\b/.test(cleanLower) || cleanLower.includes("cultura")) {
        return "Dirección de Cultura (DICU)";
      }
      if (/\b(dicyht|diep)\b/.test(cleanLower) || cleanLower.includes("investigacion cientifica")) {
        return "Dirección de Investigación Científica (DICYHT)";
      }
      if (/\b(seapi)\b/.test(cleanLower) || cleanLower.includes("proyectos de infraestructura")) {
        return "Secretaría Ejecutiva de Proyectos de Infraestructura (SEAPI)";
      }
      if (/\b(dircom)\b/.test(cleanLower) || cleanLower.includes("comunicacion estrategica")) {
        return "Dirección de Comunicación Estratégica (DIRCOM)";
      }
      
      return s.length <= 6 && !s.includes(" ") ? s.toUpperCase() : formatSpanishText(s);
    }

    // Extractor flexible de campos soportando múltiples sinónimos y regex
    function findRowField(rawRow, normRow, patterns, fallback = "") {
      for (const p of patterns) {
        if (typeof p === "string" && rawRow[p] !== undefined && String(rawRow[p]).trim() !== "") {
          return String(rawRow[p]).trim();
        }
      }
      for (const p of patterns) {
        if (typeof p === "string") {
          const k = stripAccents(p).toLowerCase().trim();
          if (normRow[k] !== undefined && String(normRow[k]).trim() !== "") {
            return String(normRow[k]).trim();
          }
        }
      }
      for (const p of patterns) {
        if (p instanceof RegExp) {
          for (const key of Object.keys(normRow)) {
            if (p.test(key) && normRow[key] !== undefined && String(normRow[key]).trim() !== "") {
              return String(normRow[key]).trim();
            }
          }
        }
      }
      return fallback;
    }

    // Formateador robusto de fechas a texto largo en español (ej: martes 19 de agosto de 2025)
    function formatDateValue(rawVal) {
      if (rawVal === undefined || rawVal === null || rawVal === '' || rawVal === '-') return '-';

      const dias = ['domingo', 'lunes', 'martes', 'miércoles', 'jueves', 'viernes', 'sábado'];
      const meses = ['enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio', 'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'];

      // Si es objeto Date
      if (rawVal instanceof Date && !isNaN(rawVal.getTime())) {
        return `${dias[rawVal.getDay()]} ${rawVal.getDate()} de ${meses[rawVal.getMonth()]} de ${rawVal.getFullYear()}`;
      }

      const str = String(rawVal).trim();

      // Si es un número serial de Excel (ej: 45889 o '45889')
      if (/^\d{5}(\.\d+)?$/.test(str)) {
        if (typeof window !== "undefined" && window.XLSX && window.XLSX.SSF && window.XLSX.SSF.parse_date_code) {
          const parsed = window.XLSX.SSF.parse_date_code(parseFloat(str));
          if (parsed && parsed.y && parsed.m && parsed.d) {
            const d = new Date(parsed.y, parsed.m - 1, parsed.d);
            return `${dias[d.getDay()]} ${parsed.d} de ${meses[parsed.m - 1]} de ${parsed.y}`;
          }
        }
        const num = parseFloat(str);
        const utcDays = Math.floor(num - 25569);
        const date = new Date(utcDays * 86400 * 1000);
        return `${dias[date.getUTCDay()]} ${date.getUTCDate()} de ${meses[date.getUTCMonth()]} de ${date.getUTCFullYear()}`;
      }

      // Si es formato DD/MM/YYYY o DD-MM-YYYY
      const ddmmyyyy = str.match(/^(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})$/);
      if (ddmmyyyy) {
        const d = parseInt(ddmmyyyy[1], 10);
        const m = parseInt(ddmmyyyy[2], 10) - 1;
        const y = parseInt(ddmmyyyy[3], 10);
        const date = new Date(y, m, d);
        if (!isNaN(date.getTime())) {
          return `${dias[date.getDay()]} ${d} de ${meses[m]} de ${y}`;
        }
      }

      // Si es formato ISO YYYY-MM-DD
      const isoMatch = str.match(/^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);
      if (isoMatch) {
        const y = parseInt(isoMatch[1], 10);
        const m = parseInt(isoMatch[2], 10) - 1;
        const d = parseInt(isoMatch[3], 10);
        const date = new Date(y, m, d);
        if (!isNaN(date.getTime())) {
          return `${dias[date.getDay()]} ${d} de ${meses[m]} de ${y}`;
        }
      }

      return str;
    }

    function detectHeaderRowAndExtract(sheet) {
      const rawMatrix = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: "" });
      if (!rawMatrix || rawMatrix.length === 0) return [];
      
      const headerKeywords = /(nombre|docente|empleado|capacitaci|curso|taller|direcci|facultad|carrera|departamento|correo|genero|sexo|accion)/i;
      
      let headerRowIndex = 0;
      let maxMatches = 0;
      
      const scanLimit = Math.min(15, rawMatrix.length);
      for (let r = 0; r < scanLimit; r++) {
        const row = rawMatrix[r];
        if (!Array.isArray(row)) continue;
        let matches = 0;
        row.forEach(cell => {
          if (cell && headerKeywords.test(String(cell))) {
            matches++;
          }
        });
        if (matches > maxMatches) {
          maxMatches = matches;
          headerRowIndex = r;
        }
      }
      
      return XLSX.utils.sheet_to_json(sheet, { range: headerRowIndex, defval: "", raw: false });
    }

        const apellidos = findRowField(row, norm, ["apellidos", "apellido", /apellido\(s\)/i, /^apellido$/i], "");

    function aggregateToUnique(data) {
      const uniqueMap = new Map();
      data.forEach(r => {
        const key = getDocenteKey(r);
        if (!uniqueMap.has(key)) {
          uniqueMap.set(key, { ...r, count: 0, Horas_Curso: 0, items: [] });
        }
        const item = uniqueMap.get(key);
        item.count += 1;
        item.Horas_Curso += Number(r.Horas_Curso) || 0;
        item.items.push({ t: r.Nombre_Capacitacion, f: r.Fecha, o: r.Fecha_Orden });
        // Preferir el N° de empleado real y un correo válido si aparecen en otra fila
        if (!isRealEmployeeId(item.ID_Docente) && isRealEmployeeId(r.ID_Docente)) item.ID_Docente = r.ID_Docente;
        if (!isValidEmail(item.Correo) && isValidEmail(r.Correo)) item.Correo = r.Correo;
      });

      return Array.from(uniqueMap.values()).map(item => {
        // Listado de capacitaciones ordenado cronológicamente
        const sorted = item.items.slice().sort((a, b) =>
          (a.o ?? Number.MAX_SAFE_INTEGER) - (b.o ?? Number.MAX_SAFE_INTEGER));
        const { items, ...rest } = item;
        return {
          ...rest,
          Nombre_Capacitacion: sorted.map((x, i) => `${i + 1}. ${x.t}`).join("\n"),
          Fecha: sorted.map((x, i) => `${i + 1}. ${x.f}`).join("\n")
        };
      });
    }

    function populateCascadeDropdowns() {
      filterConfig.forEach(target => {
        const selectEl = document.getElementById(target.id);
        const currentValue = selectEl.value || "ALL";

        const validRows = rawData.filter(row => {
          return filterConfig.every(other => {
            if (other.id === target.id) return true;
            const otherVal = document.getElementById(other.id).value;
            return !otherVal || otherVal === "ALL" || String(row[other.field]) === otherVal;
          });
        });

        const isUniqueMode = document.getElementById("filterViewMode").value === "unico";
        const countMap = new Map();
        validRows.forEach(r => {
          const v = r[target.field];
          if (!v) return;
          if (!countMap.has(v)) countMap.set(v, isUniqueMode ? new Set() : 0);
          if (isUniqueMode) countMap.get(v).add(getDocenteKey(r));
          else countMap.set(v, countMap.get(v) + 1);
        });
        const countOf = v => isUniqueMode ? countMap.get(v).size : countMap.get(v);

        let uniqueValues = [...countMap.keys()];
        if (target.field === "Mes") {
          const monthOrder = buildMonthOrder(validRows);
          uniqueValues.sort((a, b) => (monthOrder.get(a) - monthOrder.get(b)) || a.localeCompare(b, 'es'));
        } else {
          uniqueValues.sort((a, b) => a.localeCompare(b, 'es'));
        }

        selectEl.innerHTML = `<option value="ALL">${target.label} (${uniqueValues.length})</option>`;
        uniqueValues.forEach(val => {
          const opt = document.createElement("option");
          opt.value = val;
          opt.textContent = `${val} (${countOf(val).toLocaleString()})`;
          selectEl.appendChild(opt);
        });

        if (uniqueValues.includes(currentValue)) {
          selectEl.value = currentValue;
        } else {
          selectEl.value = "ALL";
        }
      });
    }

        return Object.values(row).some(v => String(v).toLowerCase().includes(searchQuery));

    function groupByCount(data, field, maxItems = null) {
      const counts = {};
      data.forEach(row => {
        const key = row[field] || "Sin especificar";
        counts[key] = (counts[key] || 0) + 1;
      });
      let entries = Object.entries(counts).sort((a, b) => b[1] - a[1]);
      const total = entries.length;
      if (maxItems && entries.length > maxItems) {
        const rest = entries.slice(maxItems);
        entries = entries.slice(0, maxItems);
        entries.push([`${OTHERS_PREFIX}${rest.length} más)`, rest.reduce((s, e) => s + e[1], 0)]);
      }
      return { labels: entries.map(e => e[0]), values: entries.map(e => e[1]), total };
    }

      if (charts[canvasId]) {
        charts[canvasId].destroy();
      }
      const ctx = document.getElementById(canvasId).getContext("2d");
      const isCircular = type === "doughnut" || type === "pie";

      charts[canvasId] = new Chart(ctx, {
        type,
        data: { labels, datasets },
        plugins: [ChartDataLabels],

          responsive: true,
          maintainAspectRatio: false,
          animation: { duration: 300 },
          layout: {
             padding: isCircular ? 0 : { right: 40 }
          },
          plugins: {
            legend: { 
              display: isCircular, 
              position: "bottom", 
              labels: { 
                boxWidth: 11, 
                font: { size: 10, family: "Helvetica Neue" },
                // Leyenda inteligente para circulares (muestra el valor entre paréntesis)
                generateLabels: (chart) => {
                  const data = chart.data;
                  if (data.labels.length && data.datasets.length) {
                    return data.labels.map((label, i) => {
                      const meta = chart.getDatasetMeta(0);
                      const style = meta.controller.getStyle(i);
                      const value = data.datasets[0].data[i];
                      return {
                        text: `${label} (${value})`,
                        fillStyle: style.backgroundColor,
                        strokeStyle: style.borderColor,
                        lineWidth: style.borderWidth,
                        hidden: !chart.getDataVisibility(i),
                        index: i
                      };
                    });
                  }
                  return [];
                }
              } 
            },
            datalabels: {
              display: !isCircular, // Oculta los números sobre las donas, solo los muestra en barras
              color: '#003366',
              anchor: 'end',
              align: 'right',
              offset: 4,
              font: { weight: 'bold', size: 11, family: "Helvetica Neue" },
              formatter: (value) => value > 0 ? value.toLocaleString() : ''
            }
          },
          scales: isCircular ? {} : {

            y: { ticks: { font: { size: 10, family: "Helvetica Neue" } } }
          },
          ...extraOptions
        }
      });

      if (filterId) attachClickFilter(charts[canvasId], filterId);
    }

        async function processExcelFiles(fileList, append = false) {
      if (!fileList || fileList.length === 0) return;

      let combinedRows = append ? [...rawData] : [];
      let filesRead = 0;
      let sheetsProcessedCount = 0;

      for (const file of fileList) {
        const arrayBuffer = await file.arrayBuffer();
        const data = new Uint8Array(arrayBuffer);
        const workbook = XLSX.read(data, { type: "array" });

        // Identificar todas las hojas que puedan contener datos estructurados
        const validSheets = [];
        workbook.SheetNames.forEach(sName => {
          const sheet = workbook.Sheets[sName];
          if (sheet && sheet["!ref"]) {
            const rows = detectHeaderRowAndExtract(sheet);
            if (rows.length > 0) {
              validSheets.push({ name: sName, rows });
            }
          }
        });

        if (validSheets.length === 0) {
          // Fallback a primera hoja
          const sheet = workbook.Sheets[workbook.SheetNames[0]];
          const jsonRows = XLSX.utils.sheet_to_json(sheet, { defval: "", raw: false });
          const mapped = jsonRows.map((row, i) => mapExcelRowToRecord(row, combinedRows.length + i));
          combinedRows = combinedRows.concat(mapped);
          sheetsProcessedCount++;
        } else {
          // Si una hoja se llama 'general', 'base' o 'consolidado', o si tienen datos similares
          const mainSheet = validSheets.find(s => 
            /general|consolidado|base|maestro|todas/i.test(s.name)
          );

          if (mainSheet && validSheets.length > 1 && mainSheet.rows.length >= 50) {
            // Se usa la hoja principal consolidada
            const mapped = mainSheet.rows.map((row, i) => mapExcelRowToRecord(row, combinedRows.length + i));
            combinedRows = combinedRows.concat(mapped);
            sheetsProcessedCount++;
          } else {
            // Procesar las hojas válidas
            validSheets.forEach(s => {
              const mapped = s.rows.map((row, i) => mapExcelRowToRecord(row, combinedRows.length + i));
              combinedRows = combinedRows.concat(mapped);
              sheetsProcessedCount++;
            });
          }
        }
        filesRead++;
      }

      if (combinedRows.length === 0) {
        alert("No se encontraron registros válidos en los archivos seleccionados.");
        return;
      }

      combinedRows = unifyCanonicalValues(combinedRows);
      assignDocenteKeys(combinedRows);

      // Deduplicación no destructiva: Solo descartar filas que sean duplicados 100% exactos en todos los datos
      const seenExact = new Set();
      let duplicatesCount = 0;
      rawData = combinedRows.filter(r => {
        const uniqueRowSignature = `${getDocenteKey(r)}|${getComparisonKey(r.Nombre_Capacitacion)}|${getComparisonKey(r.Fecha)}|${getComparisonKey(r.Direccion)}|${r.Horas_Curso}`;
        if (seenExact.has(uniqueRowSignature)) {
          duplicatesCount++;
          return false;
        }
        seenExact.add(uniqueRowSignature);
        return true;
      });

      const uniqueTotal = new Set(rawData.map(r => getDocenteKey(r))).size;
      const dupInfo = duplicatesCount > 0 ? ` (${duplicatesCount} duplicados exactos)` : '';
      document.getElementById("loadedFilesInfo").textContent =
        `Archivos cargados: ${filesRead} (hojas: ${sheetsProcessedCount}) | Participaciones: ${rawData.length.toLocaleString()}${dupInfo} | Docentes únicos: ${uniqueTotal.toLocaleString()}`;
      saveSession();

      resetFilters();
    }

    // --- (1) Clave única del docente: N° Empleado -> Correo -> Nombre normalizado ---
    function getDocenteKey(r) {
      return r.Docente_Key || r.ID_Docente || r.Nombre_Docente;
    }

    function isRealEmployeeId(id) {
      return !!id && !/^REG-/i.test(String(id));
    }

    function isValidEmail(mail) {
      return !!mail && mail !== "-" && /@/.test(mail);
    }

    function assignDocenteKeys(records) {
      const emailToKey = new Map();
      const nameToKey = new Map();

      // Primero indexar a los docentes que SÍ tienen número de empleado
      records.forEach(r => {
        if (!isRealEmployeeId(r.ID_Docente)) return;
        const key = "ID:" + String(r.ID_Docente).trim();
        if (isValidEmail(r.Correo) && !emailToKey.has(r.Correo)) emailToKey.set(r.Correo, key);
        const nk = getComparisonKey(r.Nombre_Docente);
        if (nk && nk !== "sin nombre registrado" && !nameToKey.has(nk)) nameToKey.set(nk, key);
      });

      // Asignar la clave a todos los registros (los que no tienen ID se enlazan por correo o nombre)
      records.forEach(r => {
        if (isRealEmployeeId(r.ID_Docente)) {
          r.Docente_Key = "ID:" + String(r.ID_Docente).trim();
          return;
        }
        const nk = getComparisonKey(r.Nombre_Docente);
        const validName = nk && nk !== "sin nombre registrado";
        if (isValidEmail(r.Correo) && emailToKey.has(r.Correo)) r.Docente_Key = emailToKey.get(r.Correo);
        else if (validName && nameToKey.has(nk)) r.Docente_Key = nameToKey.get(nk);
        else if (isValidEmail(r.Correo)) r.Docente_Key = "MAIL:" + r.Correo;
        else if (validName) r.Docente_Key = "NOM:" + nk;
        else r.Docente_Key = "ROW:" + r.ID_Docente;
      });
      return records;
    }

    // --- (2) Información de fecha: Mes/Año y orden cronológico ---
    const MESES_ES = ["enero", "febrero", "marzo", "abril", "mayo", "junio", "julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre"];

    function monthIndexFromName(name) {
      const n = String(name || "").toLowerCase();
      if (n === "setiembre") return 8;
      return MESES_ES.indexOf(n);
    }

    function getDateInfo(str) {
      const raw = cleanString(str);
      if (!raw || raw === "-") return { mes: "Sin fecha", orden: null };
      const s = stripAccents(raw).toLowerCase();

      // "martes 19 de agosto de 2025" / "19 y 21 de agosto de 2025"
      let m = s.match(/(\d{1,2})\s+de\s+([a-z]+)\s+(?:de|del)\s+(\d{4})/);
      if (m) {
        const mi = monthIndexFromName(m[2]);
        if (mi >= 0) {
          const y = parseInt(m[3], 10);
          return { mes: `${MESES_ES[mi].charAt(0).toUpperCase() + MESES_ES[mi].slice(1)} ${y}`, orden: new Date(y, mi, parseInt(m[1], 10)).getTime() };
        }
      }
      // "agosto 2025" / "agosto de 2025"
      m = s.match(/\b([a-z]+)\s+(?:de\s+|del\s+)?(\d{4})\b/);
      if (m) {
        const mi = monthIndexFromName(m[1]);
        if (mi >= 0) {
          const y = parseInt(m[2], 10);
          return { mes: `${MESES_ES[mi].charAt(0).toUpperCase() + MESES_ES[mi].slice(1)} ${y}`, orden: new Date(y, mi, 1).getTime() };
        }
      }
      return { mes: formatSpanishText(raw, "Sin fecha"), orden: null };
    }

    function buildMonthOrder(rows) {
      const order = new Map();
      rows.forEach(r => {
        const k = r.Mes || "Sin fecha";
        const o = typeof r.Fecha_Orden === "number" ? r.Fecha_Orden : Number.MAX_SAFE_INTEGER;
        if (!order.has(k) || o < order.get(k)) order.set(k, o);
      });
      return order;
    }

    function escapeHtml(val) {
      return String(val ?? "").replace(/[&<>"']/g, c => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
    }

    // --- (4) Excel: ancho automático de columnas y hoja de Resumen ---
    function setAutoWidth(ws, aoa) {
      const widths = [];
      aoa.forEach(row => {
        (row || []).forEach((cell, ci) => {
          const longestLine = String(cell ?? "").split("\n").reduce((mx, line) => Math.max(mx, line.length), 0);
          widths[ci] = Math.max(widths[ci] || 8, Math.min(longestLine + 2, 70));
        });
      });
      ws["!cols"] = widths.map(w => ({ wch: w }));
    }

    function getActiveFiltersText() {
      const parts = [];
      filterConfig.forEach(cfg => {
        const v = document.getElementById(cfg.id)?.value;
        if (v && v !== "ALL") parts.push(`${cfg.label.replace(/^Tod[oa]s (los|las) /i, "")}: ${v}`);
      });
      const q = document.getElementById("searchInput")?.value.trim();
      if (q) parts.push(`Búsqueda: "${q}"`);
      return parts.join(" | ");
    }

    function buildSummarySheet(data) {
      const aoa = [];
      const sumHoras = data.reduce((s, r) => s + (Number(r.Horas_Curso) || 0), 0);
      const docentes = new Set(data.map(getDocenteKey));
      const fem = data.filter(r => r.Genero === "Femenino");
      const masc = data.filter(r => r.Genero === "Masculino");

      aoa.push(["RESUMEN EJECUTIVO - DASHBOARD DOCENTES IPSD UNAH"]);
      aoa.push(["Generado", new Date().toLocaleString("es-HN")]);
      aoa.push(["Filtros aplicados", getActiveFiltersText() || "Ninguno (todos los registros cargados)"]);
      aoa.push([]);
      aoa.push(["INDICADOR", "VALOR"]);
      aoa.push(["Total participaciones (cupos)", data.length]);
      aoa.push(["Total docentes únicos", docentes.size]);
      aoa.push(["Acciones formativas distintas", new Set(data.map(r => r.Nombre_Capacitacion)).size]);
      aoa.push(["Horas acumuladas", sumHoras]);
      aoa.push(["Promedio de capacitaciones por docente", docentes.size ? +(data.length / docentes.size).toFixed(2) : 0]);
      aoa.push(["Promedio de horas por docente", docentes.size ? +(sumHoras / docentes.size).toFixed(2) : 0]);
      aoa.push(["Participaciones femeninas", fem.length]);
      aoa.push(["Docentes femeninas únicas", new Set(fem.map(getDocenteKey)).size]);
      aoa.push(["Participaciones masculinas", masc.length]);
      aoa.push(["Docentes masculinos únicos", new Set(masc.map(getDocenteKey)).size]);

      const monthOrder = buildMonthOrder(data);
      const sections = [
        ["Dirección", "Direccion"], ["Centro Regional", "Centro_Regional"], ["Facultad", "Facultad"],
        ["Carrera", "Carrera"], ["Departamento", "Departamento"], ["Tipo de Contratación", "Tipo_Contratacion"],
        ["Jornada", "Jornada"], ["Género", "Genero"], ["Mes / Año", "Mes"], ["Acción Formativa", "Nombre_Capacitacion"]
      ];

      sections.forEach(([title, field]) => {
        const map = new Map();
        data.forEach(r => {
          const k = r[field] || "Sin especificar";
          if (!map.has(k)) map.set(k, { p: 0, d: new Set(), h: 0 });
          const m = map.get(k);
          m.p++;
          m.d.add(getDocenteKey(r));
          m.h += Number(r.Horas_Curso) || 0;
        });
        let entries = [...map.entries()];
        if (field === "Mes") entries.sort((a, b) => (monthOrder.get(a[0]) ?? 0) - (monthOrder.get(b[0]) ?? 0));
        else entries.sort((a, b) => b[1].p - a[1].p);

        aoa.push([]);
        aoa.push([title.toUpperCase(), "Participaciones", "Docentes únicos", "Horas", "% Participaciones"]);
        entries.forEach(([k, m]) => {
          aoa.push([k, m.p, m.d.size, m.h, data.length ? +((m.p / data.length) * 100).toFixed(1) : 0]);
        });
      });

      const ws = XLSX.utils.aoa_to_sheet(aoa);
      setAutoWidth(ws, aoa);
      return ws;
    }

    // --- (5) Persistencia local en el navegador (IndexedDB) ---
    const IDB_NAME = "dashboardDocentesIPSD";
    const IDB_STORE = "sesion";

    function idbOpen() {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(IDB_NAME, 1);
        req.onupgradeneeded = () => req.result.createObjectStore(IDB_STORE);
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
      });
    }

    async function idbRun(mode, fn) {
      const db = await idbOpen();
      return new Promise((resolve, reject) => {
        const tx = db.transaction(IDB_STORE, mode);
        const req = fn(tx.objectStore(IDB_STORE));
        tx.oncomplete = () => { db.close(); resolve(req ? req.result : undefined); };
        tx.onerror = () => { db.close(); reject(tx.error); };
      });
    }

    async function saveSession() {
      try {
        const info = document.getElementById("loadedFilesInfo").textContent;
        await idbRun("readwrite", store => store.put({ rawData, info, savedAt: Date.now() }, "data"));
      } catch (err) {
        console.warn("No se pudo guardar la sesión en el navegador:", err);
      }
    }

    async function restoreSession() {
      try {
        const saved = await idbRun("readonly", store => store.get("data"));
        if (!saved || !Array.isArray(saved.rawData) || saved.rawData.length === 0) return;
        rawData = saved.rawData;
        rawData.forEach(r => {
          if (!r.Mes) {
            const info = getDateInfo(r.Fecha);
            r.Mes = info.mes;
            r.Fecha_Orden = info.orden;
          }
        });
        if (rawData.some(r => !r.Docente_Key)) assignDocenteKeys(rawData);
        const fecha = new Date(saved.savedAt || Date.now()).toLocaleString("es-HN");
        document.getElementById("loadedFilesInfo").textContent = `${saved.info} · Restaurado (${fecha})`;
        resetFilters();
      } catch (err) {
        console.warn("No se pudo restaurar la sesión guardada:", err);
      }
    }

    async function clearStoredData() {
      if (!confirm("¿Deseas borrar los archivos cargados y los datos guardados en este navegador?")) return;
      try {
        await idbRun("readwrite", store => store.delete("data"));
      } catch (err) {
        console.warn("No se pudo borrar la sesión guardada:", err);
      }
      rawData = [];
      document.getElementById("loadedFilesInfo").textContent = "Sin archivos cargados — Sube uno o varios archivos Excel";
      resetFilters();
    }
